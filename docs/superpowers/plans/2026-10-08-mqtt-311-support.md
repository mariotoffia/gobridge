# MQTT 3.1.1 support — implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The `mqtt` transport can talk to MQTT 3.1.1 brokers when
`options.session.protocol_version: v3.1.1` is set. MQTT 5 stays the default.

**Architecture:** A `net.Conn` wrapper (`acl_mqtt311_conn.go`) sits below the
existing pre-decode guard and translates MQTT 5 ⇄ MQTT 3.1.1 packets. paho,
the guard and every session behaviour keep speaking MQTT 5. Several pieces are
added around the translator:
- config validation for what 3.1.1 cannot express;
- a startup warning;
- an egress shortcut that skips properties;
- a durable-identity part, appended on v3.1.1 only;
- short-lived-connection damping.

**Spec:** `docs/superpowers/specs/2026-10-08-mqtt-311-support-design.md` (read
it first, §3–§5 are normative). Issue: #109.

**Tech stack:** Go 1.25, `github.com/eclipse/paho.golang` v0.23.0 (`packets`
only), testify, Mosquitto 2.1.2 via `testutil/mqttlocal`.

package paho

import (
	"fmt"

	"github.com/mariotoffia/gobridge/domain/connectivity"
	"github.com/mariotoffia/gobridge/domain/shared"
)

// validateProtocol rejects the options the session's protocol version cannot
// express. MQTT 5 accepts every option. On MQTT 3.1.1 it rejects:
//
//   - no_local: 3.1.1 has no No-Local, and loop prevention must not silently
//     disappear (ADR 0010);
//   - a non-zero session_expiry_interval: 3.1.1 cannot send one, the broker
//     decides how long the session lives;
//   - a password without a username (MQTT-3.1.2-22);
//   - clean_start: true on a persistent session, which has no 3.1.1 wire form.
//
// mode is empty when the caller does not know the session mode yet; the
// mode-dependent rule then waits for ValidateEffectiveSession. Callers must pass
// options as configured, before NewSession's defaults are applied.
func (o SessionOptions) validateProtocol(mode connectivity.SessionMode) error {
	switch o.ProtocolVersion {
	case "", ProtocolVersion5:
		return nil
	case ProtocolVersion311:
	default:
		return shared.ErrInvalidConfig.WithMessage(fmt.Sprintf(
			"mqtt: session.protocol_version must be %q or %q, got %q",
			ProtocolVersion5, ProtocolVersion311, o.ProtocolVersion))
	}
	if o.NoLocal {
		return errUnavailableOnMQTT311("session.no_local", "MQTT 3.1.1 has no No-Local; remove it, or use v5")
	}
	if o.SessionExpiryInterval != 0 {
		return errUnavailableOnMQTT311("session.session_expiry_interval",
			"MQTT 3.1.1 cannot send a session expiry, the broker decides how long the session lives; remove it, or use v5")
	}
	if err := credentialsExpressibleOnMQTT311(o.Username, !o.Password.IsZero()); err != nil {
		return err
	}
	if mode == connectivity.SessionPersistent && o.CleanStart {
		return errUnavailableOnMQTT311("session.clean_start",
			"clean_start: true on a persistent session has no MQTT 3.1.1 form; remove it, or use v5")
	}
	return nil
}

// credentialsExpressibleOnMQTT311 rejects a password without a username, which
// MQTT 3.1.1 forbids (MQTT-3.1.2-22). Live credential rotation calls it on the
// candidate before it changes anything.
func credentialsExpressibleOnMQTT311(username string, hasPassword bool) error {
	if hasPassword && username == "" {
		return errUnavailableOnMQTT311("session.password",
			"MQTT 3.1.1 forbids a password without a username; set session.username, or use v5")
	}
	return nil
}

func errUnavailableOnMQTT311(key, why string) error {
	return shared.ErrInvalidConfig.WithMessage(fmt.Sprintf(
		"mqtt: %s is not available on protocol_version %s (%s)", key, ProtocolVersion311, why))
}
```

package paho

import (
	"bytes"
	"errors"
	"fmt"
	"io"
	"math"
	"net"
	"sync"

	"github.com/eclipse/paho.golang/packets"

// mqtt311ProtocolLevel is the CONNECT protocol level of MQTT 3.1.1 (§3.1.2.2).
const mqtt311ProtocolLevel = 4

// mqtt311Conn lets a session speak MQTT 3.1.1 while Paho, the pre-decode guard
// and every session behaviour above them keep speaking MQTT 5 (ADR 0022). It
// sits directly on the dialled stream, below the guard:
//
//	paho ⇄ mqttIngressConn ⇄ mqtt311Conn ⇄ TCP / TLS / WebSocket
//
// Writes arrive as MQTT 5 packets and leave as MQTT 3.1.1; reads arrive as
// MQTT 3.1.1 and are handed up as MQTT 5. What 3.1.1 cannot carry (properties,
// reason codes, subscription options other than QoS) is dropped on the way
// out, and what it does not send is synthesised on the way in (an empty
// property block, one UNSUBACK Success per filter). A packet that would change
// meaning in translation fails the write instead of being weakened.
type mqtt311Conn struct {
	net.Conn

	// maxPayloadBytes is the session's max_payload_bytes. An inbound PUBLISH
	// keeps at most one byte more than it, so the router's payload cap acks and
	// drops an oversized one instead of the guard dropping the connection.
	maxPayloadBytes uint32
	// maximumPacketSize bounds every other inbound packet. It is the guard's
	// ceiling, wirePacketSizeFor(max_payload_bytes).
	maximumPacketSize uint32
	// receiveMaximum is the session's receive_maximum: the inbound QoS 1/2
	// window an MQTT 5 broker would honour, enforced here because 3.1.1 cannot
	// announce it.
	receiveMaximum int
	onViolation    func(error)

	// pending holds the bytes of the outbound packet Paho is still writing.
	// Paho serialises packet writes through the guard's lock, so the bytes of
	// two packets never interleave here.
	pending []byte

	// Read side, used only by Paho's reader goroutine through the guard.
	packet  []byte
	offset  int
	discard int64
	readErr error

	// mu guards the state both directions share.
	mu sync.Mutex
	// inflight holds the packet identifiers of inbound QoS 1/2 publishes handed
	// to Paho and not yet released by its PUBACK (QoS 1) or PUBCOMP (QoS 2).
	inflight map[uint16]struct{}
	// unsubscribe remembers how many filters each outbound UNSUBSCRIBE carried:
	// a 3.1.1 UNSUBACK has no per-filter reason codes, and Paho expects one each.
	unsubscribe map[uint16]int
}

func newMQTT311Conn(
	conn net.Conn,
	maxPayloadBytes, maximumPacketSize uint32,
	receiveMaximum uint16,
	onViolation func(error),
) *mqtt311Conn {
	window := int(receiveMaximum)
	if window == 0 {
		window = math.MaxUint16
	}
	return &mqtt311Conn{
		Conn:              conn,
		maxPayloadBytes:   maxPayloadBytes,
		maximumPacketSize: maximumPacketSize,
		receiveMaximum:    window,
		onViolation:       onViolation,
		inflight:          make(map[uint16]struct{}),
		unsubscribe:       make(map[uint16]int),
	}
}

// Write translates every complete MQTT 5 packet in p to MQTT 3.1.1 and writes
// it to the broker. A partial packet waits for the rest of its bytes.
func (c *mqtt311Conn) Write(p []byte) (int, error) {
	c.pending = append(c.pending, p...)
	for {
		size, err := mqttFramedSize(c.pending)
		if err != nil {
			c.pending = nil
			return 0, err
		}
		if size == 0 || size > len(c.pending) {
			return len(p), nil
		}
		out, err := c.translateOutbound(c.pending[:size])
		c.pending = append(c.pending[:0], c.pending[size:]...)
		if err != nil {
			return 0, err
		}
		if len(out) > 0 {
			if _, err := c.Conn.Write(out); err != nil {
				return 0, err //nolint:wrapcheck // net.Conn Write must preserve the transport error.
			}
		}
	}
}

// mqttFramedSize returns the total size of the MQTT packet at the start of b,
// or 0 when b does not yet hold its whole fixed header.
func mqttFramedSize(b []byte) (int, error) {
	if len(b) < 2 {
		return 0, nil
	}
	remaining, width, err := readMQTTVBIFromBytes(b[1:])
	if errors.Is(err, io.ErrUnexpectedEOF) {
		return 0, nil
	}
	if err != nil {
		return 0, fmt.Errorf("mqtt 3.1.1: outbound packet length: %w", err)
	}
	return 1 + width + remaining, nil
}

func (c *mqtt311Conn) translateOutbound(raw []byte) ([]byte, error) {
	cp, err := packets.ReadPacket(bytes.NewReader(raw))
	if err != nil {
		return nil, fmt.Errorf("mqtt 3.1.1: decode outbound packet: %w", err)
	}
	switch p := cp.Content.(type) {
	case *packets.Connect:
		return encodeConnect311(p)
	case *packets.Publish:
		return encodePublish311(p)
	case *packets.Puback:
		c.releaseInbound(p.PacketID)
		return encodeAck311(packets.PUBACK<<4, p.PacketID), nil
	case *packets.Pubrec:
		return encodeAck311(packets.PUBREC<<4, p.PacketID), nil
	case *packets.Pubrel:
		return encodeAck311(packets.PUBREL<<4|0x02, p.PacketID), nil
	case *packets.Pubcomp:
		c.releaseInbound(p.PacketID)
		return encodeAck311(packets.PUBCOMP<<4, p.PacketID), nil
	case *packets.Subscribe:
		return encodeSubscribe311(p)
	case *packets.Unsubscribe:
		c.mu.Lock()
		c.unsubscribe[p.PacketID] = len(p.Topics)
		c.mu.Unlock()
		return encodeUnsubscribe311(p), nil
	case *packets.Pingreq:
		return []byte{packets.PINGREQ << 4, 0}, nil
	case *packets.Disconnect:
		// Only a normal disconnect is sent. Any other reason precedes a close,
		// and a 3.1.1 broker publishes the Last Will on a close without
		// DISCONNECT, as a v5 broker does for a non-zero reason (MQTT-3.1.2-8).
		if p.ReasonCode == 0 {
			return []byte{packets.DISCONNECT << 4, 0}, nil
		}
		return nil, nil
	default:
		return nil, fmt.Errorf("mqtt 3.1.1: %s has no MQTT 3.1.1 form", cp.PacketType())
	}
}

// encodeConnect311 writes a 3.1.1 CONNECT. Clean Session follows the session
// expiry: without one the v5 session ends with the connection, which is what
// Clean Session means; with one the session must outlive it.
func encodeConnect311(p *packets.Connect) ([]byte, error) {
	var expiry uint32
	if p.Properties != nil && p.Properties.SessionExpiryInterval != nil {
		expiry = *p.Properties.SessionExpiryInterval
	}
	if p.CleanStart && expiry > 0 {
		return nil, errors.New("mqtt 3.1.1: a clean start that keeps the session has no MQTT 3.1.1 form")
	}
	if p.PasswordFlag && !p.UsernameFlag {
		return nil, errors.New("mqtt 3.1.1: a password requires a username")
	}
	var flags byte
	if expiry == 0 {
		flags |= 0x02
	}
	if p.WillFlag {
		flags |= 0x04 | (p.WillQOS&0x03)<<3
		if p.WillRetain {
			flags |= 0x20
		}
	}
	if p.PasswordFlag {
		flags |= 0x40
	}
	if p.UsernameFlag {
		flags |= 0x80
	}
	body := appendMQTTBytes(nil, []byte("MQTT"))
	body = append(body, mqtt311ProtocolLevel, flags, byte(p.KeepAlive>>8), byte(p.KeepAlive))
	body = appendMQTTBytes(body, []byte(p.ClientID))
	if p.WillFlag {
		body = appendMQTTBytes(body, []byte(p.WillTopic))
		body = appendMQTTBytes(body, p.WillMessage)
	}
	if p.UsernameFlag {
		body = appendMQTTBytes(body, []byte(p.Username))
	}
	if p.PasswordFlag {
		body = appendMQTTBytes(body, p.Password)
	}
	return frameMQTT311(packets.CONNECT<<4, body), nil
}

func encodePublish311(p *packets.Publish) ([]byte, error) {
	if p.Topic == "" {
		return nil, errors.New("mqtt 3.1.1: a PUBLISH needs a topic name; topic aliases have no MQTT 3.1.1 form")
	}
	header := packets.PUBLISH<<4 | (p.QoS&0x03)<<1
	if p.Duplicate && p.QoS > 0 {
		header |= 0x08
	}
	if p.Retain {
		header |= 0x01
	}
	body := appendMQTTBytes(make([]byte, 0, 4+len(p.Topic)+len(p.Payload)), []byte(p.Topic))
	if p.QoS > 0 {
		body = append(body, byte(p.PacketID>>8), byte(p.PacketID))
	}
	return frameMQTT311(header, append(body, p.Payload...)), nil
}

func encodeAck311(header byte, packetID uint16) []byte {
	return []byte{header, 2, byte(packetID >> 8), byte(packetID)}
}

// encodeSubscribe311 keeps each filter's requested QoS. Retain Handling has no
// 3.1.1 form and is dropped (every SUBSCRIBE replays retained messages); No
// Local and Retain As Published would change delivery, so they fail instead.
func encodeSubscribe311(p *packets.Subscribe) ([]byte, error) {
	body := []byte{byte(p.PacketID >> 8), byte(p.PacketID)}
	for _, sub := range p.Subscriptions {
		if sub.NoLocal || sub.RetainAsPublished {
			return nil, errors.New("mqtt 3.1.1: No Local and Retain As Published have no MQTT 3.1.1 form")
		}
		body = appendMQTTBytes(body, []byte(sub.Topic))
		body = append(body, sub.QoS&0x03)
	}
	return frameMQTT311(packets.SUBSCRIBE<<4|0x02, body), nil
}

func encodeUnsubscribe311(p *packets.Unsubscribe) []byte {
	body := []byte{byte(p.PacketID >> 8), byte(p.PacketID)}
	for _, topic := range p.Topics {
		body = appendMQTTBytes(body, []byte(topic))
	}
	return frameMQTT311(packets.UNSUBSCRIBE<<4|0x02, body)
}

func frameMQTT311(header byte, body []byte) []byte {
	out := make([]byte, 0, 5+len(body))
	out = append(out, header)
	out = appendMQTTVBI(out, len(body))
	return append(out, body...)
}

// appendMQTTBytes appends b with its two-byte length prefix (MQTT §1.5.3).
func appendMQTTBytes(dst, b []byte) []byte {
	dst = append(dst, byte(len(b)>>8), byte(len(b)))
	return append(dst, b...)
}

func (c *mqtt311Conn) releaseInbound(packetID uint16) {
	c.mu.Lock()
	delete(c.inflight, packetID)
	c.mu.Unlock()
}

// mqtt311ConnackReasonCodes maps a 3.1.1 CONNACK return code (§3.2.2.3) to the
// nearest MQTT 5 reason code, so the existing CONNACK classification applies.
var mqtt311ConnackReasonCodes = [...]byte{
	0: 0x00, // accepted
	1: 0x84, // unacceptable protocol version
	2: 0x85, // identifier rejected
	3: 0x88, // server unavailable
	4: 0x86, // bad user name or password
	5: 0x87, // not authorized
}

type inboundAdmission uint8

const (
	inboundAdmitted inboundAdmission = iota
	inboundRetransmitted
	inboundIDInUse
	inboundWindowExceeded
)

// Read hands up the next inbound packet as MQTT 5, one packet at a time.
func (c *mqtt311Conn) Read(p []byte) (int, error) {
	if len(p) == 0 {
		return 0, nil
	}
	for c.offset == len(c.packet) {
		if c.readErr != nil {
			return 0, c.readErr
		}
		packet, err := c.readPacket()
		if err != nil {
			return 0, err
		}
		c.packet, c.offset = packet, 0
	}
	n := copy(p, c.packet[c.offset:])
	c.offset += n
	return n, nil
}

// readPacket returns the next packet translated to MQTT 5, or nil for a
// retransmission that must not reach Paho.
func (c *mqtt311Conn) readPacket() ([]byte, error) {
	if c.discard > 0 {
		n := c.discard
		c.discard = 0
		if _, err := io.CopyN(io.Discard, c.Conn, n); err != nil {
			return nil, fmt.Errorf("mqtt 3.1.1: drain an oversized PUBLISH: %w", err)
		}
	}
	var fixedHeader [1]byte
	if _, err := io.ReadFull(c.Conn, fixedHeader[:]); err != nil {
		return nil, err //nolint:wrapcheck // net.Conn Read must preserve the transport error.
	}
	var scratch [4]byte
	remaining, width, err := readMQTTVBI(c.Conn, scratch[:])
	if err != nil {
		if errors.Is(err, errMQTTVBINonCanonical) || errors.Is(err, errMQTTVBITooLong) {
			return nil, c.violation(newMQTTMalformedError())
		}
		return nil, err
	}
	packetType, flags := fixedHeader[0]>>4, fixedHeader[0]&0x0F
	if packetType == packets.PUBLISH {
		return c.readPublish(flags, remaining)
	}
	if size := uint64(1 + width + remaining); c.maximumPacketSize > 0 && size > uint64(c.maximumPacketSize) {
		return nil, c.violation(&mqttIngressError{
			kind:   mqttIngressPacketTooLarge,
			actual: size,
			limit:  uint64(c.maximumPacketSize),
			cause:  shared.ErrPayloadTooLarge,
		})
	}
	body := make([]byte, remaining)
	if _, err := io.ReadFull(c.Conn, body); err != nil {
		return nil, err //nolint:wrapcheck // net.Conn Read must preserve the transport error.
	}
	packet, ok := c.translateInbound(packetType, flags, body)
	if !ok {
		return nil, c.violation(newMQTTMalformedError())
	}
	return packet, nil
}

// translateInbound checks one non-PUBLISH 3.1.1 packet exactly — Paho's
// MQTT 5 decoder would accept bytes 3.1.1 forbids, such as a reason code on a
// PUBACK — and returns its MQTT 5 form.
func (c *mqtt311Conn) translateInbound(packetType, flags byte, body []byte) ([]byte, bool) {
	switch packetType {
	case packets.CONNACK:
		if flags != 0 || len(body) != 2 || body[0]&^0x01 != 0 ||
			int(body[1]) >= len(mqtt311ConnackReasonCodes) || (body[0] == 1 && body[1] != 0) {
			return nil, false
		}
		return []byte{packets.CONNACK << 4, 3, body[0], mqtt311ConnackReasonCodes[body[1]], 0}, true
	case packets.PUBACK, packets.PUBREC, packets.PUBREL, packets.PUBCOMP:
		var want byte
		if packetType == packets.PUBREL {
			want = 0x02
		}
		if flags != want || len(body) != 2 || (body[0] == 0 && body[1] == 0) {
			return nil, false
		}
		return []byte{packetType<<4 | flags, 2, body[0], body[1]}, true
	case packets.SUBACK:
		if flags != 0 || len(body) < 3 || (body[0] == 0 && body[1] == 0) {
			return nil, false
		}
		for _, code := range body[2:] {
			if code > 2 && code != 0x80 {
				return nil, false
			}
		}
		return frameMQTT311(packets.SUBACK<<4, append([]byte{body[0], body[1], 0}, body[2:]...)), true
	case packets.UNSUBACK:
		if flags != 0 || len(body) != 2 {
			return nil, false
		}
		packetID := uint16(body[0])<<8 | uint16(body[1])
		c.mu.Lock()
		filters, known := c.unsubscribe[packetID]
		delete(c.unsubscribe, packetID)
		c.mu.Unlock()
		if packetID == 0 || !known {
			return nil, false
		}
		// Packet identifier, an empty property block, then one Success per filter.
		v5 := make([]byte, 3+filters)
		v5[0], v5[1] = body[0], body[1]
		return frameMQTT311(packets.UNSUBACK<<4, v5), true
	case packets.PINGRESP:
		if flags != 0 || len(body) != 0 {
			return nil, false
		}
		return []byte{packets.PINGRESP << 4, 0}, true
	default:
		return nil, false
	}
}

// readPublish translates one inbound PUBLISH. It never buffers more payload
// than one byte past max_payload_bytes: an oversized PUBLISH is handed up at
// once with its payload cut there, so the router acks and drops it in receive
// order, and the rest is discarded at the start of the next read. Handing the
// head up first lets the ack reach the broker while a slow drain is still
// running.
func (c *mqtt311Conn) readPublish(flags byte, remaining int) ([]byte, error) {
	qos := (flags >> 1) & 0x03
	duplicate := flags&0x08 != 0
	if qos == 3 || (qos == 0 && duplicate) || remaining < 2 {
		return nil, c.violation(newMQTTMalformedError())
	}
	var topicLength [2]byte
	if _, err := io.ReadFull(c.Conn, topicLength[:]); err != nil {
		return nil, err //nolint:wrapcheck // net.Conn Read must preserve the transport error.
	}
	topicSize := int(topicLength[0])<<8 | int(topicLength[1])
	header := 2 + topicSize // topic, then the packet identifier for QoS 1/2
	if qos > 0 {
		header += 2
	}
	if topicSize == 0 || header > remaining {
		return nil, c.violation(newMQTTMalformedError())
	}
	payloadSize := remaining - header
	keep := min(payloadSize, int(c.maxPayloadBytes)+1)

	v5Remaining := header + 1 + keep // plus the empty property block
	packet := make([]byte, 0, 5+v5Remaining)
	packet = append(packet, packets.PUBLISH<<4|flags)
	packet = appendMQTTVBI(packet, v5Remaining)
	packet = append(packet, topicLength[0], topicLength[1])
	rest := len(packet)
	packet = packet[:rest+header-2]
	if _, err := io.ReadFull(c.Conn, packet[rest:]); err != nil {
		return nil, err //nolint:wrapcheck // net.Conn Read must preserve the transport error.
	}
	if bytes.ContainsAny(packet[rest:rest+topicSize], "+#\x00") {
		return nil, c.violation(newMQTTMalformedError())
	}
	if qos > 0 {
		packetID := uint16(packet[len(packet)-2])<<8 | uint16(packet[len(packet)-1])
		if packetID == 0 {
			return nil, c.violation(newMQTTMalformedError())
		}
		switch c.admitInbound(packetID, duplicate) {
		case inboundRetransmitted:
			// MQTT 5 forbids resending on a live connection; 3.1.1 allows it.
			// The copy Paho already holds settles both, so this one is skipped:
			// handed up, its late ack could release a newer message that
			// reused the identifier.
			c.discard = int64(payloadSize)
			return nil, nil
		case inboundIDInUse:
			return nil, c.violation(newMQTTMalformedError())
		case inboundWindowExceeded:
			return nil, c.violation(&mqttIngressError{
				kind:  mqttIngressWindowExceeded,
				limit: uint64(c.receiveMaximum),
				cause: shared.ErrProtocolError,
			})
		}
	}
	packet = append(packet, 0) // MQTT 3.1.1 carries no properties
	payloadStart := len(packet)
	packet = packet[:payloadStart+keep]
	if _, err := io.ReadFull(c.Conn, packet[payloadStart:]); err != nil {
		return nil, err //nolint:wrapcheck // net.Conn Read must preserve the transport error.
	}
	c.discard = int64(payloadSize - keep)
	return packet, nil
}

// admitInbound records an inbound QoS 1/2 packet identifier. A broker that
// resends one still in flight is retransmitting (DUP set) or broken (DUP
// clear); one that exceeds receive_maximum would block Paho's reader, which a
// closed socket does not release, so it is refused before Paho sees it.
func (c *mqtt311Conn) admitInbound(packetID uint16, duplicate bool) inboundAdmission {
	c.mu.Lock()
	defer c.mu.Unlock()
	if _, inFlight := c.inflight[packetID]; inFlight {
		if duplicate {
			return inboundRetransmitted
		}
		return inboundIDInUse
	}
	if len(c.inflight) >= c.receiveMaximum {
		return inboundWindowExceeded
	}
	c.inflight[packetID] = struct{}{}
	return inboundAdmitted
}

// violation reports a packet the session must not accept and closes the socket
// without a DISCONNECT: a 3.1.1 broker then publishes the Last Will, as a v5
// broker does after the guard's reject reason codes.
func (c *mqtt311Conn) violation(err error) error {
	c.readErr = err
	if c.onViolation != nil {
		c.onViolation(err)
	}
	_ = c.Conn.Close()
	return err
}

// noteMQTT311ConnectionLifetimeLocked reports whether a connection that just
// went down on MQTT 3.1.1 was short-lived: up for less than
// connectionStabilityWindow, and not dropped by this session's own ingress
// reject. MQTT 3.1.1 has no Session-Taken-Over reason code, so a client-id
// collision shows only as connections that keep dying soon after they come up.
// A connection that had been stable ends any such storm. Callers hold s.mu.
func (s *Session) noteMQTT311ConnectionLifetimeLocked(now int64) bool {
	if !s.opts.protocolV311() || s.connUpAt == 0 {
		return false
	}
	if now-s.connUpAt >= int64(connectionStabilityWindow) {
		s.takeoverStreak = 0
		return false
	}
	return s.lastIngressRejectAt < s.connUpAt
}

// noteShortLivedMQTT311Connection damps the reconnect storm two instances
// sharing a client_id cause on MQTT 3.1.1. It shares takeoverStreak and
// takeoverPenalty with noteSessionTakeover: the first drop costs nothing, as a
// legitimate exclusive failover, and each further one doubles the penalty. It
// counts no MetricMQTTSessionTakeover, because the cause is inferred, not
// reported.
func (s *Session) noteShortLivedMQTT311Connection() {
	now := s.clock().Now().UnixNano()
	s.mu.Lock()
	s.takeoverStreak++
	s.lastTakeoverAt = now
	streak := s.takeoverStreak
	s.mu.Unlock()
	if streak >= 3 && s.logger != nil {
		s.logger.Error("mqtt: the MQTT 3.1.1 connection keeps dropping soon after it connects; reconnects are being backed off. "+
			"On MQTT 3.1.1 another live instance using the same client_id looks exactly like this",
			"client_id", s.opts.ClientID,
			"drop_count", streak,
			"backoff_penalty", s.takeoverPenalty(),
		)
	}
}
```
